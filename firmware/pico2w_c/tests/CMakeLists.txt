cmake_minimum_required(VERSION 3.16)
project(clock_host_tests C)
set(CMAKE_C_STANDARD 11)
enable_testing()
if(MSVC)
    add_compile_options(/W4 /WX /utf-8 /UNDEBUG)
    add_compile_definitions(_CRT_SECURE_NO_WARNINGS)
else()
    add_compile_options(-Wall -Wextra -Werror -UNDEBUG)
endif()
set(APP "${CMAKE_CURRENT_LIST_DIR}/../app")
set(SERVICES "${CMAKE_CURRENT_LIST_DIR}/../services")
set(BT "${CMAKE_CURRENT_LIST_DIR}/../../../experiments/bluetooth_a2dp_nest/nest_autotest")
include_directories("${APP}" "${SERVICES}" "${BT}")
add_executable(alarm_engine_test alarm_engine_host_test.c "${APP}/alarm_engine.c" "${APP}/clock_time.c")
add_executable(clock_time_test clock_time_test.c "${APP}/clock_time.c")
add_executable(bt_catalog_test bt_catalog_test.c "${SERVICES}/bt_catalog.c")
add_executable(bt_policy_test bt_policy_test.c)
add_executable(timer_warning_test timer_warning_test.c "${APP}/timer_warning.c")
add_executable(timer_controls_test timer_controls_test.c "${APP}/timer_controls.c" "${APP}/countdown.c")
add_executable(music_catalog_test music_catalog_test.c "${APP}/music_catalog.c")
add_executable(sd_catalog_test sd_catalog_test.c "${APP}/music_catalog.c")
add_executable(music_playlist_test music_playlist_test.c "${APP}/music_playlist.c")
add_executable(birthday_test birthday_test.c "${APP}/birthday.c" "${APP}/clock_time.c")
add_executable(music_sleep_test music_sleep_test.c "${APP}/music_sleep.c")
add_executable(audio_queue_test audio_queue_test.c "${CMAKE_CURRENT_LIST_DIR}/../audio/queue.c")
add_executable(speech_plan_test speech_plan_test.c "${CMAKE_CURRENT_LIST_DIR}/../audio/speech.c"
    "${CMAKE_CURRENT_LIST_DIR}/../audio/queue.c" "${APP}/clock_time.c")
target_include_directories(speech_plan_test PRIVATE "${CMAKE_CURRENT_LIST_DIR}/../audio" "${CMAKE_CURRENT_LIST_DIR}/../drivers")
target_include_directories(audio_queue_test PRIVATE "${CMAKE_CURRENT_LIST_DIR}/../audio" "${CMAKE_CURRENT_LIST_DIR}/../drivers")
add_executable(settings_store_test settings_store_test.c "${APP}/alarm_engine.c" "${APP}/clock_time.c" "${APP}/birthday.c")
add_executable(lcd_cursor_test lcd_cursor_test.c "${CMAKE_CURRENT_LIST_DIR}/../drivers/lcd1602_i2c.c")
add_executable(phone_access_test phone_access_test.c "${SERVICES}/phone_access.c"
    "${APP}/alarm_engine.c" "${APP}/birthday.c" "${APP}/clock_time.c")
add_executable(network_time_test network_time_test.c "${APP}/clock_time.c")
add_executable(phone_overview_test phone_overview_test.c "${SERVICES}/phone_overview.c"
    "${APP}/alarm_engine.c" "${APP}/clock_time.c")
add_executable(phone_control_test phone_control_test.c "${SERVICES}/phone_control.c" "${APP}/alarm_engine.c" "${APP}/clock_time.c")
target_include_directories(network_time_test PRIVATE network_mocks)
target_compile_definitions(network_time_test PRIVATE CLOCK_ENABLE_PHONE_WEB=1)
target_include_directories(lcd_cursor_test PRIVATE mocks "${CMAKE_CURRENT_LIST_DIR}/../drivers")
target_include_directories(settings_store_test PRIVATE mocks)
foreach(test timer_warning music_catalog sd_catalog alarm_engine clock_time bt_catalog bt_policy settings_store timer_controls music_playlist music_sleep audio_queue speech_plan birthday lcd_cursor phone_access network_time phone_overview phone_control)
    add_test(NAME ${test} COMMAND ${test}_test)
endforeach()
add_test(NAME git_version COMMAND "${CMAKE_COMMAND}"
    "-DSOURCE_DIR=${CMAKE_CURRENT_LIST_DIR}/.."
    -P "${CMAKE_CURRENT_LIST_DIR}/git_version_test.cmake")
find_program(NODE_EXECUTABLE node)
if(NODE_EXECUTABLE)
    add_test(NAME phone_sliders COMMAND "${NODE_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/phone_sliders_test.js")
    add_test(NAME phone_timer_wheels COMMAND "${NODE_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/phone_timer_wheels_test.js")
endif()
