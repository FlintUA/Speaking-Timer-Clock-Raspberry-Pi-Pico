cmake_minimum_required(VERSION 3.16)
project(clock_host_tests C)
set(CMAKE_C_STANDARD 11)
enable_testing()
if(MSVC)
    add_compile_options(/W4 /WX /utf-8 /UNDEBUG)
    add_compile_definitions(_CRT_SECURE_NO_WARNINGS)
else()
    add_compile_options(-Wall -Wextra -Werror -UNDEBUG)
endif()
set(APP "${CMAKE_CURRENT_LIST_DIR}/../app")
set(SERVICES "${CMAKE_CURRENT_LIST_DIR}/../services")
set(BT "${CMAKE_CURRENT_LIST_DIR}/../../../experiments/bluetooth_a2dp_nest/nest_autotest")
include_directories("${APP}" "${SERVICES}" "${BT}")
add_executable(alarm_engine_test alarm_engine_host_test.c "${APP}/alarm_engine.c")
add_executable(clock_time_test clock_time_test.c "${APP}/clock_time.c")
add_executable(bt_catalog_test bt_catalog_test.c "${SERVICES}/bt_catalog.c")
add_executable(settings_store_test settings_store_test.c "${APP}/alarm_engine.c")
target_include_directories(settings_store_test PRIVATE mocks)
foreach(test alarm_engine clock_time bt_catalog settings_store)
    add_test(NAME ${test} COMMAND ${test}_test)
endforeach()
